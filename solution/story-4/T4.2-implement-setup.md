```xml

    <id>testcontainers</id>
    <properties>
        <skip.containers>true</skip.containers>
        <skip.aggregators>true</skip.aggregators>
        <skip.integration.tests>false</skip.integration.tests>
    </properties>
    <build>
        <plugins>
            <plugin>
                <groupId>org.apache.maven.plugins</groupId>
                <artifactId>maven-failsafe-plugin</artifactId>
                <version>${maven-failsafe-plugin.version}</version>
                <configuration>
                    <includes>
                        <include>**/*TcIntegrationTest</include>
                    </includes>
                    <skip>false</skip>
                </configuration>
            </plugin>
        </plugins>
    </build>
</profile>
```

**Usage:** `mvn verify -Ptestcontainers`

This runs only Testcontainers-based tests (matching `*TcIntegrationTest` pattern) without starting any docker-compose services.

## 3. Base Test Configuration Class

```java
package uk.gov.ho.dacc.fdp.testcontainers;

import org.testcontainers.containers.GenericContainer;
import org.testcontainers.containers.KafkaContainer;
import org.testcontainers.containers.Network;
import org.testcontainers.containers.wait.strategy.Wait;
import org.testcontainers.utility.DockerImageName;

/**
 * Shared Testcontainers infrastructure for FDP integration tests.
 * <p>
 * Each test class gets isolated containers — no shared state between test classes.
 * Within a test class, containers are started once and shared across test methods.
 */
public final class TestInfrastructure {

    private TestInfrastructure() {}

    // Use consistent versions matching docker-compose.yml
    public static final String REDIS_IMAGE = "redis:5.0.6";
    public static final String KAFKA_IMAGE = "confluentinc/cp-kafka:7.5.5";
    public static final String SCHEMA_REGISTRY_IMAGE = "confluentinc/cp-schema-registry:7.5.5";
    public static final String LOCALSTACK_IMAGE = "localstack/localstack:0.12.18";

    /**
     * Create a Redis container.
     */
    public static GenericContainer<?> redis() {
        return new GenericContainer<>(DockerImageName.parse(REDIS_IMAGE))
                .withExposedPorts(6379)
                .waitingFor(Wait.forLogMessage(".*Ready to accept connections.*", 1));
    }

    /**
     * Create a Kafka container (includes ZooKeeper internally).
     * Note: docker-compose.yml uses ZooKeeper-based Kafka (cp-kafka:7.5.5 + cp-zookeeper:7.5.5).
     * Testcontainers KafkaContainer can use KRaft mode for simplicity in tests.
     * Both produce compatible Kafka behaviour for integration testing.
     */
    public static KafkaContainer kafka() {
        return new KafkaContainer(DockerImageName.parse(KAFKA_IMAGE));
    }

    /**
     * Create a Kafka container on a shared network (for Schema Registry).
     */
    public static KafkaContainer kafka(Network network) {
        return new KafkaContainer(DockerImageName.parse(KAFKA_IMAGE))
                .withNetwork(network)
                .withNetworkAliases("kafka");
    }

    /**
     * Create a Schema Registry container connected to a Kafka container.
     */
    public static GenericContainer<?> schemaRegistry(Network network, KafkaContainer kafka) {
        return new GenericContainer<>(DockerImageName.parse(SCHEMA_REGISTRY_IMAGE))
                .withNetwork(network)
                .withExposedPorts(8081)
                .withEnv("SCHEMA_REGISTRY_HOST_NAME", "schema-registry")
                .withEnv("SCHEMA_REGISTRY_KAFKASTORE_BOOTSTRAP_SERVERS",
                         "PLAINTEXT://kafka:9092")
                .dependsOn(kafka)
                .waitingFor(Wait.forHttp("/subjects").forStatusCode(200));
    }

    /**
     * Create a LocalStack container with IAM service.
     */
    public static GenericContainer<?> localstack() {
        return new GenericContainer<>(DockerImageName.parse(LOCALSTACK_IMAGE))
                .withExposedPorts(4566)
                .withEnv("LOCALSTACK_SERVICES", "iam")
                .waitingFor(Wait.forHttp("/health").forStatusCode(200));
    }
}
```

## 4. Redis Integration Test Example

```java
package uk.gov.ho.dacc.fdp.integration;

import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;
import org.testcontainers.containers.GenericContainer;
import org.testcontainers.junit.jupiter.Container;
import org.testcontainers.junit.jupiter.Testcontainers;
import redis.clients.jedis.Jedis;
import uk.gov.ho.dacc.fdp.testcontainers.TestInfrastructure;

import static org.assertj.core.api.Assertions.assertThat;

/**
 * Example Testcontainers-based integration test using isolated Redis.
 * <p>
 * Run with: mvn verify -Ptestcontainers
 * 
 * Each test class gets a fresh Redis instance — no shared state.
 */
@Testcontainers
class RedisCacheTcIntegrationTest {

    @Container
    static GenericContainer<?> redis = TestInfrastructure.redis();

    private Jedis jedis;

    @BeforeEach
    void setUp() {
        jedis = new Jedis(redis.getHost(), redis.getMappedPort(6379));
        jedis.flushAll(); // Clean state per test method
    }

    @Test
    void shouldStoreAndRetrieveFromCache() {
        jedis.set("test-key", "test-value");
        
        String result = jedis.get("test-key");
        
        assertThat(result).isEqualTo("test-value");
    }

    @Test
    void shouldHaveIsolatedState() {
        // This test runs with empty Redis — no leakage from other test classes
        assertThat(jedis.dbSize()).isZero();
    }
}
```

## 5. Kafka + Schema Registry Integration Test Example

package uk.gov.ho.dacc.fdp.integration;

import org.junit.jupiter.api.Test;
import org.testcontainers.containers.GenericContainer;
import org.testcontainers.containers.KafkaContainer;
import org.testcontainers.containers.Network;
import org.testcontainers.junit.jupiter.Container;
import org.testcontainers.junit.jupiter.Testcontainers;
import uk.gov.ho.dacc.fdp.testcontainers.TestInfrastructure;

import static org.assertj.core.api.Assertions.assertThat;

/**
 * Testcontainers-based test with Kafka + Schema Registry.
 * <p>
 * Each test class gets its own Kafka cluster — topics are isolated.
 */
@Testcontainers
class KafkaMessagingTcIntegrationTest {

    static Network network = Network.newNetwork();

    @Container
    static KafkaContainer kafka = TestInfrastructure.kafka(network);

    @Container
    static GenericContainer<?> schemaRegistry = 
            TestInfrastructure.schemaRegistry(network, kafka);

    @Test
    void kafkaShouldBeRunning() {
        assertThat(kafka.isRunning()).isTrue();
    }

    @Test
    void schemaRegistryShouldBeAccessible() {
        String url = String.format("http://%s:%d", 
                schemaRegistry.getHost(), 
                schemaRegistry.getMappedPort(8081));
        assertThat(url).isNotEmpty();
        // TODO: Register schema and verify
    }
}
